launching:

source install/local_setup.bash

ros2 launch platooning_pkg spawn_platoon_FDI_3/4car.launch.py
Here write 3 or 4, not 3/4

damaging the leader:
ros2 run platooning_pkg FDI_attack --ros-args -p target_name:=tb3_leader

follower1-leader
ros2 run platooning_pkg FDI_feasable_full_graph --ros-args -p my_name:=tb3_follower1 -p target_name:=tb3_leader -p my_offset:=2.0 -p target_offset:=3.0

follower1-leader
follower1 is connected to the damaged sensor of the leader
ros2 run platooning_pkg FDI_feasable_full_graph --ros-args -p my_name:=tb3_follower1 -p target_name:=tb3_leader -p my_offset:=2.0 -p target_offset:=3.0 --remap /tb3_leader/odom:=/tb3_leader/odom_spoofed

follower2-follower1
ros2 run platooning_pkg FDI_feasable_full_graph --ros-args -p my_name:=tb3_follower2 -p target_name:=tb3_follower1 -p my_offset:=1.0 -p target_offset:=2.0

follower2-follower1
follower2 is connected to the damaged sensor of the follower1
ros2 run platooning_pkg FDI_feasable_full_graph --ros-args -p my_name:=tb3_follower2 -p target_name:=tb3_follower1 -p my_offset:=1.0 -p target_offset:=2.0 --remap /tb3_follower1/odom:=/tb3_follower1/odom_spoofed
